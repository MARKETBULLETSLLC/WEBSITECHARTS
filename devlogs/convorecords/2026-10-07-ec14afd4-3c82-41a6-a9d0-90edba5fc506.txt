var lbIndex = 0;
var lbItems = [];

function openLightbox(src, title, items, index){
  lbItems  = items  || [{ src:src, title:title }];
  lbIndex  = index !== undefined ? index : 0;
  showLb(lbIndex);
  document.getElementById('lb').classList.add('open');
  document.body.style.overflow = 'hidden';
}
function showLb(i){
  var it = lbItems[i];
  document.getElementById('lb-img').src   = it.src;
  document.getElementById('lb-title').textContent = it.title || '';
  var prev = document.getElementById('lb-prev');
  var next = document.getElementById('lb-next');
  prev.style.visibility = lbItems.length > 1 ? 'visible' : 'hidden';
  next.style.visibility = lbItems.length > 1 ? 'visible' : 'hidden';
}
function closeLightbox(){
  document.getElementById('lb').classList.remove('open');
  document.getElementById('lb-img').src = '';
  document.body.style.overflow = '';
  lbItems = []; lbIndex = 0;
}
document.getElementById('lb-close').addEventListener('click', closeLightbox);
document.getElementById('lb').addEventListener('click', function(e){
  if(e.target === this) closeLightbox();
});
document.getElementById('lb-prev').addEventListener('click', function(e){

function buildCard(file, title, sub){
  var url = img(file);

function renderNavigator(){

  NAV_CATEGORIES.forEach(function(cat, ci){

          grid.appendChild(card);
        });

        panel.appendChild(sg);

      var grid = document.createElement('div');

      cat.charts.forEach(function(c, i){

        grid.appendChild(card);
      });

    }

  tabsEl.addEventListener('click', function(e){
    var btn = e.target.closest('.nav-tab');

    if(fp) fp.classList.add('active');
  }
