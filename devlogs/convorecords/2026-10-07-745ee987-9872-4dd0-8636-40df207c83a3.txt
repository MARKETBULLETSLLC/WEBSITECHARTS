var lbIndex = 0;
var lbItems = [];

function openLightbox(src, title, items, index){
  lbItems  = items  || [{ src:src, title:title }];
  lbIndex  = index !== undefined ? index : 0;
  showLb(lbIndex);
  document.getElementById('lb').classList.add('open');
  document.body.style.overflow = 'hidden';
}
function showLb(i){
  var it = lbItems[i];
  document.getElementById('lb-img').src   = it.src;
  document.getElementById('lb-title').textContent = it.title || '';
  var prev = document.getElementById('lb-prev');
  var next = document.getElementById('lb-next');
  prev.style.visibility = lbItems.length > 1 ? 'visible' : 'hidden';
  next.style.visibility = lbItems.length > 1 ? 'visible' : 'hidden';
}
function closeLightbox(){
  document.getElementById('lb').classList.remove('open');
  document.getElementById('lb-img').src = '';
  document.body.style.overflow = '';
  lbItems = []; lbIndex = 0;
}
document.getElementById('lb-close').addEventListener('click', closeLightbox);
document.getElementById('lb').addEventListener('click', function(e){
  if(e.target === this) closeLightbox();
});
document.getElementById('lb-prev').addEventListener('click', function(e){
  e.stopPropagation();
  lbIndex = (lbIndex - 1 + lbItems.length) % lbItems.length;
  showLb(lbIndex);
});
document.getElementById('lb-next').addEventListener('click', function(e){
  e.stopPropagation();
  lbIndex = (lbIndex + 1) % lbItems.length;
  showLb(lbIndex);
});
document.addEventListener('keydown', function(e){
  if(!document.getElementById('lb').classList.contains('open')) return;
  if(e.key==='Escape') closeLightbox();
  if(e.key==='ArrowLeft'  && lbItems.length>1){ lbIndex=(lbIndex-1+lbItems.length)%lbItems.length; showLb(lbIndex); }

function buildCard(file, title, sub){
  var url = img(file);
  var card = document.createElement('div');
  card.className = 'chart-card';

  card.dataset.title = title;

    card.classList.add('no-image');
    card.innerHTML = '';
    var inner = document.createElement('div');
    inner.className = 'chart-no-img-inner';
    inner.innerHTML = '<span>&#8987;</span><span>COMING SOON</span><span>' + file + '</span>';

  tabsEl.addEventListener('click', function(e){

  }

â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â• */
var lbIndex = 0;
var lbItems = [];

function openLightbox(src, title, items, index){
  lbItems  = items  || [{ src:src, title:title }];
  lbIndex  = index !== undefined ? index : 0;
  showLb(lbIndex);
  document.getElementById('lb').classList.add('open');
  document.body.style.overflow = 'hidden';
}
function showLb(i){
  var it = lbItems[i];
  document.getElementById('lb-img').src   = it.src;
  document.getElementById('lb-title').textContent = it.title || '';
  var prev = document.getElementById('lb-prev');
  var next = document.getElementById('lb-next');
  prev.style.visibility = lbItems.length > 1 ? 'visible' : 'hidden';
  next.style.visibility = lbItems.length > 1 ? 'visible' : 'hidden';
}
function closeLightbox(){
  document.getElementById('lb').classList.remove('open');
  document.getElementById('lb-img').src = '';
  document.body.style.overflow = '';
  lbItems = []; lbIndex = 0;
}
document.getElementById('lb-close').addEventListener('click', closeLightbox);
document.getElementById('lb').addEventListener('click', function(e){
  if(e.target === this) closeLightbox();
});
document.getElementById('lb-prev').addEventListener('click', function(e){
  e.stopPropagation();
  lbIndex = (lbIndex - 1 + lbItems.length) % lbItems.length;
  showLb(lbIndex);
});
document.getElementById('lb-next').addEventListener('click', function(e){
  e.stopPropagation();
  lbIndex = (lbIndex + 1) % lbItems.length;
  showLb(lbIndex);
});
document.addEventListener('keydown', function(e){
  if(!document.getElementById('lb').classList.contains('open')) return;
  if(e.key==='Escape') closeLightbox();
  if(e.key==='ArrowLeft'  && lbItems.length>1){ lbIndex=(lbIndex-1+lbItems.length)%lbItems.length; showLb(lbIndex); }
  if(e.key==='ArrowRight' && lbItems.length>1){ lbIndex=(lbIndex+1)%lbItems.length; showLb(lbIndex); }
});

/* â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•

â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â• */
function buildCard(file, title, sub){
  var url = img(file);
  var card = document.createElement('div');
  card.className = 'chart-card';
  card.dataset.src   = url;
  card.dataset.title = title;

  var image = document.createElement('img');
  image.alt     = title;
  image.loading = 'lazy';
  image.src     = url;
  image.onerror = function(){
    card.classList.add('no-image');
    card.innerHTML = '';
    var inner = document.createElement('div');
    inner.className = 'chart-no-img-inner';
    inner.innerHTML = '<span>&#8987;</span><span>COMING SOON</span><span>' + file + '</span>';
    var cap = document.createElement('div');
    cap.className = 'chart-card-cap';
    cap.innerHTML = '<div class="chart-card-title">'+title+'</div><div class="chart-card-sub">'+sub+'</div>';
    card.appendChild(inner);
    card.appendChild(cap);
    card.style.cursor = 'default';
  };
  card.appendChild(image);

  var cap = document.createElement('div');
  cap.className = 'chart-card-cap';
  cap.innerHTML = '<div class="chart-card-title">'+title+'</div><div class="chart-card-sub">'+sub+'</div>';
  card.appendChild(cap);

  return card;
}

function processShortcodes(html){
  return html.replace(/\[chart:([^\]]+)\]/g, function(_, filename){
    var url = img(filename.trim());
    return '<a href="'+url+'" class="inline-chart-link lb-trigger" data-src="'+url+'" data-title="'+filename.replace('.png','')+'">'
      + '<img src="'+url+'" class="inline-chart-thumb" alt="'+filename+'" loading="lazy">'
      + '</a>';
  });
}
function renderCommentary(){
  var cfg = MB_CONFIG;
  document.getElementById('dateline').textContent = cfg.dateline;

  var ul = document.getElementById('bullets-list');
  ul.innerHTML = '';
  cfg.bullets.forEach(function(b){
    var li = document.createElement('li');
    li.className = 'bullet-item';
    li.innerHTML = '<div class="bullet-mark" style="background:'+b.color+';box-shadow:0 0 6px 2px '+b.color+'40;"></div>'
      + '<span style="color:'+b.color+'">'+b.text+'</span>';
    ul.appendChild(li);
  });

  document.getElementById('commentary-body').innerHTML = processShortcodes(cfg.body);
}

function renderNavigator(){
  var tabsEl   = document.getElementById('nav-tabs');
  var panelsEl = document.getElementById('nav-panels');
  var firstTab = null;

  NAV_CATEGORIES.forEach(function(cat, ci){
    // Tab button
    var btn = document.createElement('button');
    btn.className = 'nav-tab';
    btn.textContent = cat.short;
    btn.dataset.cat = cat.id;
    tabsEl.appendChild(btn);
    if(ci === 0) firstTab = btn;

    // Panel
    var panel = document.createElement('div');
    panel.className = 'nav-panel';
    panel.id = 'panel-' + cat.id;

    if(cat.groups){
      // Category with sub-groups (Spreads)
      cat.groups.forEach(function(grp){
        var sg = document.createElement('div');
        sg.className = 'nav-subgroup';
        var lbl = document.createElement('div');
        lbl.className = 'nav-subgroup-label';
        lbl.textContent = grp.label;
        sg.appendChild(lbl);
        var grid = document.createElement('div');
        grid.className = 'nav-grid';
        var lbSet = grp.charts.map(function(c){ return { src: img(c.file), title: c.title }; });
        grp.charts.forEach(function(c, i){
          var card = buildCard(c.file, c.title, c.sub);
          card.addEventListener('click', function(){ openLightbox(null, null, lbSet, i); });
          grid.appendChild(card);
        });
        sg.appendChild(grid);
        panel.appendChild(sg);
      });
    } else {
      // Simple category
      var grid = document.createElement('div');
      grid.className = 'nav-grid';
      var lbSet = cat.charts.map(function(c){ return { src: img(c.file), title: c.title }; });
      cat.charts.forEach(function(c, i){
        var card = buildCard(c.file, c.title, c.sub);
        card.addEventListener('click', function(){ openLightbox(null, null, lbSet, i); });
        grid.appendChild(card);
      });
      panel.appendChild(grid);
    }

    panelsEl.appendChild(panel);
  });

  // Tab click handlers
  tabsEl.addEventListener('click', function(e){
    var btn = e.target.closest('.nav-tab');
    if(!btn) return;
    tabsEl.querySelectorAll('.nav-tab').forEach(function(t){ t.classList.remove('active'); });
    panelsEl.querySelectorAll('.nav-panel').forEach(function(p){ p.classList.remove('active'); });
    btn.classList.add('active');
    var panel = document.getElementById('panel-' + btn.dataset.cat);
    if(panel) panel.classList.add('active');
  });

  // Activate first tab
  if(firstTab){
    firstTab.classList.add('active');
    var fp = document.getElementById('panel-' + firstTab.dataset.cat);
    if(fp) fp.classList.add('active');
  }
}
